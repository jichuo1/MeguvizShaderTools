cmake_minimum_required(VERSION 3.24)
project(MgstExternalConsumer NONE)
find_package(MeguvizShaderTools CONFIG REQUIRED)
get_target_property(mgst_path MeguvizShaderTools::mgst IMPORTED_LOCATION_RELEASE)
if(NOT mgst_path)
  get_target_property(mgst_path MeguvizShaderTools::mgst IMPORTED_LOCATION_DEBUG)
endif()
execute_process(COMMAND "${mgst_path}" --version-json OUTPUT_VARIABLE version_json COMMAND_ERROR_IS_FATAL ANY)
string(JSON lock_hash GET "${version_json}" result toolchain_lock_hash)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/inputs")
configure_file(source.comp "${CMAKE_CURRENT_BINARY_DIR}/inputs/source.comp" COPYONLY)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/source.comp" source_hash)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/inputs/request.json" "{\"schema_version\":1,\"logical_shader_id\":\"external/identity\",\"owner\":{\"repository\":\"external-consumer\",\"revision\":\"fixture-1\"},\"source\":{\"path\":\"source.comp\",\"sha256\":\"${source_hash}\"},\"includes\":[],\"profile_id\":\"native-vulkan\",\"profile_version\":\"1\",\"entry_point\":\"main\",\"target_env\":\"vulkan1.0\",\"effective_options\":{},\"numeric_profile\":\"reference-fp32\",\"toolchain_lock_hash\":\"${lock_hash}\",\"required_flags\":[],\"optimization_passes\":[],\"resource_recipe\":\"glslang-default-pinned\"}\n")
mgst_compile_shaderpack(REQUEST "${CMAKE_CURRENT_BINARY_DIR}/inputs/request.json" OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/identity-pack" OUT_PACK identity_pack DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/source.comp")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/current-pack.txt" CONTENT "${identity_pack}\n")
add_custom_target(shaderpack ALL DEPENDS "${identity_pack}/manifest.json")
add_custom_target(verify COMMAND MeguvizShaderTools::mgst verify --pack "${identity_pack}" DEPENDS shaderpack VERBATIM)
