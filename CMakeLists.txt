function(mgst_pin name revision)
  execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${MGST_DEPS_ROOT}/${name}" -C "${MGST_DEPS_ROOT}/${name}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual STREQUAL revision)
    message(FATAL_ERROR "${name}: wrong source pin ${actual}")
  endif()
  execute_process(COMMAND "${GIT_EXECUTABLE}" -c "safe.directory=${MGST_DEPS_ROOT}/${name}" -C "${MGST_DEPS_ROOT}/${name}" diff --quiet HEAD RESULT_VARIABLE dirty)
  if(NOT dirty EQUAL 0)
    message(FATAL_ERROR "${name}: tracked source differs from its locked revision")
  endif()
endfunction()
